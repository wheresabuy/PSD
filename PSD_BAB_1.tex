\subsection{Dari Fenomena Fisik Menjadi Sinyal Listrik (Peran Sensor \& Transduser)}

Dunia fisis beroperasi berdasarkan besaran-besaran alamiah non-listrik yang bersifat kontinu terhadap ruang dan waktu. Untuk menjembatani dunia analog fisis dengan sistem komputasi terotomatisasi, diperlukan perangkat \textbf{transduser} (elemen pengubah bentuk energi) dan \textbf{sensor} (elemen pendeteksi kuantitas fisik spesifik).

\begin{penurunanbox}[Asal-Usul Pemodelan Fungsi Transduksi Sensor via Deret Taylor]
Bagaimana persamaan kurva sensor $v(t) = \mathcal{S} \cdot P(t) + \beta \cdot P^2(t) + \eta(t)$ terbentuk?
\begin{enumerate}[leftmargin=*]
  \item \textbf{Fondasi Dasar:} Respon tegangan sensor $v$ terhadap besaran fisik $P$ secara umum merupakan fungsi non-linier sembarang $v = f(P)$.
  \item \textbf{Ekspansi Deret Taylor di Sekitar Titik Kerja $P_0$:}
  \[
  f(P) = f(P_0) + \left.\frac{df}{dP}\right|_{P_0}(P - P_0) + \frac{1}{2!}\left.\frac{d^2f}{dP^2}\right|_{P_0}(P - P_0)^2 + \frac{1}{3!}\left.\frac{d^3f}{dP^3}\right|_{P_0}(P - P_0)^3 + \dots
  \]
  \item \textbf{Pemisahan Komponen:}
  \begin{itemize}
    \item $f(P_0) = V_0$ adalah tegangan offset DC (dikalibrasi menjadi 0).
    \item Koefisien linier $\mathcal{S} = \left.\frac{df}{dP}\right|_{P_0}$ disebut \textbf{Sensitivitas Nominal Sensor} (Volt/Satuan Fisis).
    \item Koefisien kuadratik $\beta = \frac{1}{2}\left.\frac{d^2f}{dP^2}\right|_{P_0}$ merepresentasikan \textbf{Deviasi Non-Linearitas Orde-2}.
  \end{itemize}
  \item \textbf{Derau Termal Acak ($\eta(t)$):} Elektron di dalam atom konduktor bergetar secara acak karena energi panas $k_B T$. Menurut **Teorema Nyquist-Johnson (1928)**, kerapatan daya derau tegangan pada resistor $R$ diturunkan dari mekanika statistik Boltzmann:
  \begin{equation}
  \overline{v_n^2} = 4 k_B T R \Delta f \quad [\text{V}^2]
  \end{equation}
  di mana $k_B = 1.380649 \times 10^{-23}\text{ J/K}$, $T$ temperatur Kelvin, $R$ resistansi Thévenin, dan $\Delta f$ bandwidth.
\end{enumerate}
\end{penurunanbox}

\begin{center}
  \includegraphics[width=0.92\linewidth]{assets/fenomena_fisik_ke_sinyal.png}
  \captionof{figure}{Konversi analitis fenomena fisis alamiah (Akustik, Termal, Seismik) menjadi fluktuasi sinyal tegangan kontinu.}
\end{center}

\begin{penurunanbox}[Asal-Usul Fisika Gelombang Akustik, Termal, dan Seismik]
\begin{enumerate}[leftmargin=*]
  \item \textbf{Gelombang Suara (Helmholtz 1-D):}
  \begin{itemize}
    \item Hukum II Newton fluida udara: $-\frac{\partial p}{\partial x} = \rho_0 \frac{\partial u}{\partial t}$ ($p$ tekanan, $u$ kecepatan partikel).
    \item Hukum Kontinuitas massa: $-\frac{\partial u}{\partial x} = \frac{1}{B}\frac{\partial p}{\partial t}$ ($B$ bulk modulus elastisitas).
    \item Turunkan persamaan pertama terhadap $x$ dan persamaan kedua terhadap $t$, lalu substitusikan $\frac{\partial^2 u}{\partial x \partial t}$:
    \begin{equation}
    \frac{\partial^2 p(x,t)}{\partial x^2} - \frac{1}{c_s^2}\frac{\partial^2 p(x,t)}{\partial t^2} = 0 \quad \text{dengan laju rambat } c_s = \sqrt{\frac{B}{\rho_0}} \approx 343\text{ m/s}
    \end{equation}
  \end{itemize}

  \item \textbf{Respon Termal Orde-1 Sensor Termokopel:}
  \begin{itemize}
    \item Hukum I Termodinamika (Kekekalan Energi): Kalor yang diserap oleh massa sensor $m$ sama dengan perpindahan panas konveksi Newton:
    \[
    m c_p \frac{dT(t)}{dt} = h A \left( T_{\text{lingkungan}} - T(t) \right)
    \]
    \item Definisikan konstanta waktu termal $\tau_{\text{th}} = \frac{m c_p}{h A} = R_{\text{th}} C_{\text{th}}$. Persamaan menjadi:
    \[
    \tau_{\text{th}}\frac{dT(t)}{dt} + T(t) = T_{\text{akhir}}
    \]
    \item Solusi persamaan diferensial homogen + partikular dengan syarat awal $T(0) = T_{\text{awal}}$:
    \begin{equation}
    T(t) = T_{\text{akhir}} + (T_{\text{awal}} - T_{\text{akhir}})e^{-t/\tau_{\text{th}}} + w_{\text{thermal}}(t)
    \end{equation}
  \end{itemize}

  \item \textbf{Osilasi Seismometer Teredam Kerak Bumi:}
  \begin{itemize}
    \item Hukum II Newton pada sensor massa-pegas-peredam di dalam rumah seismometer yang bergetar $\ddot{x}_g(t)$:
    \[
    m \ddot{x}(t) + c \dot{x}(t) + k x(t) = -m \ddot{x}_g(t)
    \]
    \item Bagi dengan massa $m$, definisikan frekuensi natural $\omega_n = \sqrt{k/m}$ dan rasio redaman $\zeta = \frac{c}{2\sqrt{km}}$:
    \[
    \ddot{x}(t) + 2\zeta\omega_n \dot{x}(t) + \omega_n^2 x(t) = -\ddot{x}_g(t)
    \]
    \item Untuk kondisi teredam kurang ($\zeta < 1$), akar persamaan karakteristiknya berbentuk kompleks terkonjugasi $s_{1,2} = -\zeta\omega_n \pm j\omega_n\sqrt{1-\zeta^2}$, menghasilkan respon transien eksponensial teredam:
    \begin{equation}
    a(t) = A_0 e^{-\zeta \omega_n (t - t_0)} \sin\left(\omega_n \sqrt{1 - \zeta^2}(t - t_0)\right) u(t - t_0)
    \end{equation}
  \end{itemize}
\end{enumerate}
\end{penurunanbox}

\subsection{Paradigma Pemrosesan: Rantai ASP (Analog) vs Rantai DSP (Digital)}

\begin{center}
  \includegraphics[width=0.92\linewidth]{assets/diagram_asp_vs_dsp.png}
  \captionof{figure}{Komparasi arsitektur perangkat keras: Pemrosesan Sinyal Analog (ASP) vs Pemrosesan Sinyal Digital (DSP).}
\end{center}

\begin{penurunanbox}[Asal-Usul Persamaan Diferensial ASP vs Persamaan Beda DSP]
\begin{enumerate}[leftmargin=*]
  \item \textbf{Asal-Usul Rantai ASP (Hukum Sirkuit Kirchhoff):}
  \begin{itemize}
    \item Pada rangkaian analog RLC, hubungan arus-tegangan komponen adalah $v_R = iR$, $v_L = L\frac{di}{dt}$, dan $i_C = C\frac{dv_C}{dt}$.
    \item Aplikasi Hukum Tegangan Kirchhoff ($\sum V = 0$) dan Hukum Arus Kirchhoff ($\sum I = 0$) menghasilkan Persamaan Diferensial Linier Koefisien Konstan (LCCDE):
    \begin{equation}
    \sum_{k=0}^N a_k \frac{d^k y(t)}{dt^k} = \sum_{m=0}^M b_m \frac{d^m x(t)}{dt^m}
    \end{equation}
    \item Transformasi Laplace $\mathcal{L}\{\frac{d^k y}{dt^k}\} = s^k Y(s)$ menghasilkan fungsi transfer rasional $H(s) = \frac{Y(s)}{X(s)} = \frac{\sum b_m s^m}{\sum a_k s^k}$.
  \end{itemize}

  \item \textbf{Asal-Usul Rantai DSP (Aproksimasi Selisih Diskrit):}
  \begin{itemize}
    \item Pada domain diskrit dengan periode sampling $T_s$, turunan waktu pertama didekati oleh selisih mundur (*backward difference*): $\frac{dy(t)}{dt} \approx \frac{y[n] - y[n-1]}{T_s}$.
    \item Turunan orde lebih tinggi didekati oleh kombinasi selisih mundur berulang. Mensubstitusikan aproksimasi diferensial ini ke LCCDE kontinu menghasilkan Persamaan Beda Linier Koefisien Konstan (LCCDDE):
    \begin{equation}
    \sum_{k=0}^N a_k y[n-k] = \sum_{m=0}^M b_m x[n-m]
    \end{equation}
    \item Transformasi-Z $\mathcal{Z}\{y[n-k]\} = z^{-k} Y(z)$ menghasilkan fungsi transfer domain-Z $H(z) = \frac{Y(z)}{X(z)} = \frac{\sum b_m z^{-m}}{\sum a_k z^{-k}}$, yang diimplementasikan secara deterministik melalui instruksi komputasi MAC (*Multiply-Accumulate*) di prosesor digital.
  \end{itemize}
\end{enumerate}
\end{penurunanbox}

\subsection{Tiga Pilar Anatomi Gelombang: Amplitudo, Frekuensi, dan Fase}

Persamaan matematis analitis dari suatu gelombang sinusoidal murni dinyatakan sebagai berikut:
\begin{equation}
x(t) = A \sin(2\pi f t + \phi) = A \sin(\Omega t + \phi)
\end{equation}

\begin{center}
  \includegraphics[width=0.85\linewidth]{assets/anatomi_sinyal.png}
  \captionof{figure}{Anatomi geometris sinyal sinusoidal kontinu: Amplitudo Puncak ($A$), Nilai Peak-to-Peak ($V_{\text{pp}} = 2A$), Periode Fundamental ($T$), dan Pergeseran Fase ($\phi$).}
\end{center}

\begin{center}
  \includegraphics[width=0.92\linewidth]{assets/komparasi_sinyal.png}
  \captionof{figure}{Efek ortogonal pemisahan parameter gelombang: Modulasi Frekuensi (Panel 1), Modulasi Amplitudo (Panel 2), dan Modulasi Fase Kuadratur (Panel 3).}
\end{center}

\begin{penurunanbox}[Penurunan Matematis Nilai RMS, Daya Sinyal, dan Pergeseran Waktu]
\begin{enumerate}[leftmargin=*]
  \item \textbf{Penurunan Daya Rata-rata $P_{\text{avg}} = A^2/2$ \& Tegangan Efektif $V_{\text{rms}} = A/\sqrt{2}$:}
  \begin{itemize}
    \item Energi disipasi pada beban $1\ \Omega$ selama satu periode $T$ adalah integral dari kuadrat tegangan sesaat:
    \[
    P_{\text{avg}} = \frac{1}{T}\int_0^T x^2(t) dt = \frac{1}{T}\int_0^T A^2 \sin^2(\Omega t + \phi) dt
    \]
    \item Gunakan identitas trigonometri sudut ganda $\sin^2\theta = \frac{1 - \cos(2\theta)}{2}$:
    \[
    P_{\text{avg}} = \frac{A^2}{2T} \left[ \int_0^T 1\ dt - \int_0^T \cos(2\Omega t + 2\phi) dt \right]
    \]
    \item Karena integral dari fungsi kosinus periodik selama satu atau lebih periode penuh bernilai nol ($\int_0^T \cos(2\Omega t + 2\phi) dt = 0$), persamaan menyusut menjadi:
    \begin{equation}
    P_{\text{avg}} = \frac{A^2}{2T} [T - 0] = \mathbf{\frac{A^2}{2}} \implies V_{\text{rms}} = \sqrt{P_{\text{avg}}} = \mathbf{\frac{A}{\sqrt{2}}} \approx 0.7071 A
    \end{equation}
  \end{itemize}

  \item \textbf{Penurunan Hubungan Fase ($\phi$) terhadap Pergeseran Waktu ($\Delta t$):}
  \begin{itemize}
    \item Faktorkan kecepatan sudut $\Omega$ dari dalam argumen sinus:
    \[
    x(t) = A \sin(\Omega t + \phi) = A \sin\left( \Omega \left( t + \frac{\phi}{\Omega} \right) \right) = A \sin(\Omega(t - \Delta t))
    \]
    \item Dengan membandingkan kedua bentuk di atas, pergeseran translasi waktu ekuivalen adalah:
    \begin{equation}
    \mathbf{\Delta t = -\frac{\phi}{\Omega} = -\frac{\phi}{2\pi f}}
    \end{equation}
    Jika $\phi > 0$, maka $\Delta t < 0$ (grafik bergeser ke kiri / mendahului waktu acuan $t=0$).
  \end{itemize}
\end{enumerate}
\end{penurunanbox}

\subsection{Sistem Pengolah Sinyal \& 4 Klasifikasi Karakteristik Operasinya}

\begin{center}
  \includegraphics[width=0.95\linewidth]{assets/klasifikasi_sistem.png}
  \captionof{figure}{Empat pilar klasifikasi sistem: (1) Linearitas, (2) Time-Invariance, (3) Kausalitas, dan (4) Stabilitas BIBO.}
\end{center}

\begin{penurunanbox}[Penurunan Formal \& Pembuktian Uji Karakteristik Sistem]
\begin{enumerate}[leftmargin=*]
  \item \textbf{Pembuktian Prinsip Superposisi Linearitas:}
  \begin{itemize}
    \item Sistem linier didefinisikan mematuhi pemetaan aditif $\mathcal{T}\{x_1 + x_2\} = \mathcal{T}\{x_1\} + \mathcal{T}\{x_2\}$ dan homogen $\mathcal{T}\{\alpha x\} = \alpha \mathcal{T}\{x\}$.
    \item \textit{Mengapa sistem $y[n] = 0.5 x^3[n]$ non-linier?} Masukkan $x[n] = \alpha x_1[n]$:
    \[
    y[n] = 0.5 (\alpha x_1[n])^3 = \alpha^3 \cdot (0.5 x_1^3[n]) = \alpha^3 y_1[n] \neq \alpha y_1[n]
    \]
    Karena muncul faktor $\alpha^3$, sifat homogenitas gagal. Jika dimasukkan $\cos(\omega_0 n)$, ekspansi kubik $\cos^3\theta = \frac{3\cos\theta + \cos(3\theta)}{4}$ membangkitkan frekuensi baru $3\omega_0$ (distorsi harmonisa).
  \end{itemize}

  \item \textbf{Penurunan Syarat Keterjumlahan Mutlak Stabilitas BIBO:}
  \begin{itemize}
    \item Misalkan input dibatasi $|x[n]| \le M_x < \infty$. Hubungan input-output sistem LTI dinyatakan oleh konvolusi $y[n] = \sum_{k=-\infty}^\infty h[k] x[n-k]$.
    \item Ambil nilai mutlak dari output $|y[n]|$ dan gunakan pertidaksamaan segitiga (*triangle inequality*):
    \[
    |y[n]| = \left| \sum_{k=-\infty}^{\infty} h[k] x[n-k] \right| \le \sum_{k=-\infty}^{\infty} |h[k] x[n-k]| = \sum_{k=-\infty}^{\infty} |h[k]| \cdot |x[n-k]|
    \]
    \item Karena $|x[n-k]| \le M_x$, keluarkan konstanta $M_x$ dari deret:
    \[
    |y[n]| \le M_x \sum_{k=-\infty}^{\infty} |h[k]|
    \]
    \item Agar $|y[n]| \le M_y < \infty$ untuk sembarang masukan terbatas $M_x$, maka syarat mutlak yang wajib dipenuhi adalah deret respon impuls harus konvergen:
    \begin{equation}
    \mathbf{S_h = \sum_{k=-\infty}^{\infty} |h[k]| < \infty \quad (\text{Absolutely Summable})}
    \end{equation}
  \end{itemize}
\end{enumerate}
\end{penurunanbox}

\subsection{Rantai 4 Tahap Lengkap Digitalisasi ADC}

\begin{center}
  \includegraphics[width=0.95\linewidth]{assets/tahapan_adc_sampling_kuantisasi.png}
  \captionof{figure}{Empat tahapan lengkap digitalisasi ADC: (1) Sinyal Analog Asli, (2) Pencuplikan Diskrit Waktu, (3) Kuantisasi Amplitudo Diskrit, dan (4) Pengkodean Biner.}
\end{center}

\begin{penurunanbox}[Asal-Usul Matematis Modulasi Deretan Impuls Dirac]
Bagaimana sinyal kontinu $x_a(t)$ dimodelkan saat dicuplik setiap interval $T_s$?
\begin{enumerate}[leftmargin=*]
  \item \textbf{Fungsi Kereta Impuls Dirac (*Dirac Comb*):} $p(t) = \sum_{n=-\infty}^\infty \delta(t - n T_s)$.
  \item \textbf{Perkalian Waktu:} Sinyal tercuplik $x_s(t) = x_a(t) \cdot p(t) = \sum_{n=-\infty}^\infty x_a(n T_s) \delta(t - n T_s)$.
  \item \textbf{Representasi Deret Fourier Kompleks dari Kereta Dirac:} Karena $p(t)$ periodik dengan periode $T_s$, ia dapat diekspansikan ke deret Fourier:
  \[
  p(t) = \sum_{k=-\infty}^{\infty} c_k e^{j 2\pi k F_s t}, \quad \text{di mana } c_k = \frac{1}{T_s}\int_{-T_s/2}^{T_s/2} \delta(t) e^{-j 2\pi k F_s t} dt = \frac{1}{T_s} = F_s
  \]
  \item Maka $p(t) = F_s \sum_{k=-\infty}^\infty e^{j 2\pi k F_s t}$.
  \item \textbf{Transformasi Fourier Perkalian Sinyal:}
  \[
  X_s(F) = \mathcal{F}\left\{ x_a(t) \cdot F_s \sum_{k=-\infty}^\infty e^{j 2\pi k F_s t} \right\} = F_s \sum_{k=-\infty}^\infty \mathcal{F}\left\{ x_a(t) e^{j 2\pi k F_s t} \right\}
  \]
  Berdasarkan sifat modulasi frekuensi $\mathcal{F}\{x_a(t) e^{j 2\pi F_0 t}\} = X_a(F - F_0)$, maka terbukti secara eksak:
  \begin{equation}
  \mathbf{X_s(F) = F_s \sum_{k=-\infty}^{\infty} X_a(F - k F_s)}
  \end{equation}
\end{enumerate}
\end{penurunanbox}

\subsection{Pencuplikan (Sampling Clock), Teorema Nyquist, dan Bencana Aliasing}

\begin{center}
  \includegraphics[width=0.95\linewidth]{assets/sampling_nyquist_aliasing.png}
  \captionof{figure}{Fenomena rekonstruksi gelombang terhadap variasi laju pencuplikan $F_s$ pada sinyal uji $f = 5\text{ Hz}$.}
\end{center}

\begin{penurunanbox}[Penurunan Rumus Lipatan Aliasing \& Rekonstruksi Whittaker-Shannon]
\begin{enumerate}[leftmargin=*]
  \item \textbf{Asal-Usul Rumus Frekuensi Alias $f_{\text{alias}} = |F - k F_s|$:}
  \begin{itemize}
    \item Spektrum $X_s(F)$ terdiri dari salinan $X_a(F)$ yang digeser sejauh kelipatan $F_s$.
    \item Filter rekonstruksi di penerima hanya membuka jendela frekuensi pada pita dasar $|F| \le F_s/2$.
    \item Jika komponen frekuensi asli $F > F_s/2$, salinan spektral pada indeks $k$ akan masuk ke dalam pita dasar. Karena sinyal riil memiliki spektrum simetris genap $|X(F)| = |X(-F)|$, frekuensi positif semu yang tertangkap adalah jarak absolut terdekat ke kelipatan sampling:
    \begin{equation}
    f_{\text{alias}} = |F - k F_s|, \quad k = \operatorname{round}(F / F_s)
    \end{equation}
  \end{itemize}

  \item \textbf{Penurunan Rumus Rekonstruksi Sinc Whittaker-Shannon:}
  \begin{itemize}
    \item Untuk memulihkan $X_a(F)$ dari $X_s(F)$, kita kalikan $X_s(F)$ dengan filter lolos-rendah ideal $H_r(F) = T_s \cdot \operatorname{rect}\left(\frac{F}{F_s}\right)$ yang bernilai $T_s$ untuk $|F| \le F_s/2$ dan $0$ untuk lainnya.
    \item Transformasi Fourier balik dari fungsi kotak $H_r(F)$ adalah fungsi sinus kardinal:
    \[
    h_r(t) = \int_{-F_s/2}^{F_s/2} T_s e^{j 2\pi F t} dF = T_s \left[ \frac{e^{j 2\pi F t}}{j 2\pi t} \right]_{-F_s/2}^{F_s/2} = T_s \frac{\sin(\pi F_s t)}{\pi t} = \operatorname{sinc}\left(\frac{t}{T_s}\right)
    \]
    \item Di domain waktu, perkalian spektrum menjadi konvolusi:
    \begin{align}
    x_a(t) &= x_s(t) * h_r(t) = \left( \sum_{n=-\infty}^\infty x[n] \delta(t - n T_s) \right) * \operatorname{sinc}\left(\frac{t}{T_s}\right) \nonumber \\
    &= \mathbf{\sum_{n=-\infty}^\infty x[n] \operatorname{sinc}\left(\frac{t - n T_s}{T_s}\right)}
    \end{align}
  \end{itemize}
\end{enumerate}
\end{penurunanbox}

\subsection{Kuantisasi \& Pengkodean Biner 3-Bit (Rentang 0 s.d. 10 Volt)}

\begin{center}
  \includegraphics[width=0.90\linewidth]{assets/kuantisasi_3bit_0_10v.png}
  \captionof{figure}{Fungsi tangga kuantisasi uniform 3-bit: Titik batas partisi input, level representasi tengah $V_q$, dan error kuantisasi maksimal $\pm \Delta/2$.}
\end{center}

\begin{penurunanbox}[Penurunan Eksak Daya Derau Kuantisasi $\sigma_e^2 = \Delta^2/12$ dan Formula SQNR]
\begin{enumerate}[leftmargin=*]
  \item \textbf{Penurunan Integral Variansi Galat Derau Kuantisasi:}
  \begin{itemize}
    \item Asumsikan sinyal melintasi banyak level kuantisasi sehingga galat pembulatan $e = x_q - x$ berdistribusi acak seragam kontinu pada interval $[-\Delta/2, +\Delta/2]$.
    \item Fungsi kerapatan probabilitas (PDF) adalah tinggi balok konstan: $f_E(e) = \frac{1}{\Delta}$.
    \item Daya derau kuantisasi adalah momen kedua (ekspektasi nilai kuadrat $e^2$):
    \begin{align}
    P_e = \sigma_e^2 = \mathbb{E}[e^2] &= \int_{-\Delta/2}^{+\Delta/2} e^2 f_E(e) de = \int_{-\Delta/2}^{+\Delta/2} e^2 \frac{1}{\Delta} de = \frac{1}{\Delta} \left[ \frac{e^3}{3} \right]_{-\Delta/2}^{+\Delta/2} \nonumber \\
    &= \frac{1}{\Delta} \left[ \frac{(\Delta/2)^3}{3} - \frac{(-\Delta/2)^3}{3} \right] = \frac{1}{\Delta} \left[ \frac{\Delta^3/8}{3} - \left(-\frac{\Delta^3/8}{3}\right) \right] \nonumber \\
    &= \frac{1}{\Delta} \left[ \frac{\Delta^3}{24} + \frac{\Delta^3}{24} \right] = \frac{1}{\Delta} \left[ \frac{2\Delta^3}{24} \right] = \mathbf{\frac{\Delta^2}{12}}
    \end{align}
  \end{itemize}

  \item \textbf{Penurunan Rumus Baku SQNR $\approx 6.02 B + 1.76\text{ dB}$:}
  \begin{itemize}
    \item Rentang dinamis total adalah $V_{\text{range}} = 2^B \Delta$.
    \item Sinyal sinusoidal penuh memiliki tegangan puncak-ke-puncak $V_{\text{pp}} = V_{\text{range}} = 2^B \Delta$, sehingga amplitudonya $A = 2^{B-1}\Delta$.
    \item Daya sinyal sinusoidal murni adalah $P_s = \frac{A^2}{2} = \frac{(2^{B-1}\Delta)^2}{2} = \frac{2^{2B-2}\Delta^2}{2} = \frac{2^{2B}\Delta^2}{8}$.
    \item Rasio daya sinyal terhadap derau (SQNR linier):
    \[
    \frac{P_s}{P_e} = \frac{\frac{2^{2B}\Delta^2}{8}}{\frac{\Delta^2}{12}} = \frac{12}{8} \cdot 2^{2B} = 1.5 \cdot 2^{2B}
    \]
    \item Konversikan ke skala logaritmik Desibel (dB):
    \begin{align}
    \text{SQNR}_{\text{dB}} &= 10 \log_{10}\left( 1.5 \cdot 2^{2B} \right) = 10 \log_{10}(1.5) + 10 \log_{10}\left( 2^{2B} \right) \nonumber \\
    &= 10(0.17609) + 20 B \log_{10}(2) = 1.7609 + 20 B (0.30103) \nonumber \\
    &= \mathbf{6.0206 \cdot B + 1.7609 \approx 6.02 \cdot B + 1.76 \quad [\text{dB}]}
    \end{align}
  \end{itemize}
\end{enumerate}
\end{penurunanbox}

\subsection{Sinyal Multikanal (Multi-Channel Signals) \& Representasi Vektor-Matriks}

\begin{center}
  \includegraphics[width=0.92\linewidth]{assets/sinyal_multikanal.png}
  \captionof{figure}{Akuisisi Sinyal Multikanal 4-kanal EKG simultan dan pemetaannya ke struktur aljabar matriks spasio-temporal $\mathbf{X}_{M \times N}$.}
\end{center}

\begin{penurunanbox}[Asal-Usul Matriks Kovariansi Spasial Multikanal]
Bagaimana persamaan matriks kovariansi spasial $\mathbf{R}_{xx} = \frac{1}{N} \mathbf{X} \mathbf{X}^T$ diturunkan dari konsep statistik dasar?
\begin{enumerate}[leftmargin=*]
  \item \textbf{Definisi Statistik Kovariansi Antar-Dua Sensor:} Kovariansi spasial antara kanal ke-$i$ ($x_i[n]$) dan kanal ke-$j$ ($x_j[n]$) dengan rata-rata nol adalah:
  \[
  R_{ij} = \frac{1}{N}\sum_{n=0}^{N-1} x_i[n] x_j[n]
  \]
  \item \textbf{Bentuk Perkalian Titik Vektor Baris:} Ambil baris ke-$i$ matriks $\mathbf{X}$, yaitu $\mathbf{x}_i^T = [x_i[0], x_i[1], \dots, x_i[N-1]]$, dan baris ke-$j$, yaitu $\mathbf{x}_j^T$. Perkalian titik $\mathbf{x}_i^T \mathbf{x}_j$ adalah $\sum_{n=0}^{N-1} x_i[n] x_j[n]$.
  \item \textbf{Sintesis Seluruh Matriks:} Dalam aljabar linier, menghitung perkalian titik untuk seluruh pasangan $i,j \in \{1, \dots, M\}$ setara secara eksak dengan mengalikan matriks data $\mathbf{X} \in \mathbb{R}^{M \times N}$ dengan transposisinya $\mathbf{X}^T \in \mathbb{R}^{N \times M}$:
  \begin{equation}
  \mathbf{R}_{xx} = \frac{1}{N} \mathbf{X} \mathbf{X}^T = \frac{1}{N} \begin{bmatrix}
  \sum x_1^2[n] & \sum x_1[n]x_2[n] & \cdots & \sum x_1[n]x_M[n] \\
  \sum x_2[n]x_1[n] & \sum x_2^2[n] & \cdots & \sum x_2[n]x_M[n] \\
  \vdots & \vdots & \ddots & \vdots \\
  \sum x_M[n]x_1[n] & \sum x_M[n]x_2[n] & \cdots & \sum x_M^2[n]
  \end{bmatrix} \in \mathbb{R}^{M \times M}
  \end{equation}
\end{enumerate}
\end{penurunanbox}

  \phi_{u,v}(x,y) = e^{j 2\pi u x} \cdot e^{j 2\pi v y} = e^{j 2\pi (ux + vy)}
  \]

  \end{equation}

\end{enumerate}

\subsection{Sinyal Waktu Diskrit (Discrete-Time Signals / DTS) \& Sinyal Elementer}

\begin{center}
  \includegraphics[width=0.95\linewidth]{assets/sinyal_waktu_diskrit.png}
  \captionof{figure}{Representasi Sinyal Waktu Diskrit (Stem Plot Sampling, Fluktuasi Diskrit Finansial) dan Tiga Sinyal Elementer Fundamental.}
\end{center}

\begin{penurunanbox}[Pembuktian Sifat Penyaringan (*Sifting Property*) Sinyal Diskrit]
Mengapa sembarang sinyal diskrit $x[n]$ selalu dapat dituliskan sebagai $x[n] = \sum_{k=-\infty}^\infty x[k]\delta[n-k]$?
\begin{enumerate}[leftmargin=*]
  \item Berdasarkan definisi impuls Kronecker:
  \[
  \delta[n - k] = \begin{cases} 1, & \text{saat } k = n \\ 0, & \text{saat } k \neq n \end{cases}
  \]
  \item Evaluasi suku-suku deret penjumlahan tak terhingga:
  \[
  \sum_{k=-\infty}^{\infty} x[k]\delta[n-k] = \dots + x[n-1]\underbrace{\delta[1]}_{0} + x[n]\underbrace{\delta[0]}_{1} + x[n+1]\underbrace{\delta[-1]}_{0} + \dots = x[n] \cdot 1 = \mathbf{x[n]}
  \]
  Setiap titik sampel $x[k]$ dipandang sebagai impuls berskala yang "ditembakkan" pada posisi indeks waktu $k$. Inilah dasar lahirnya rumus **Konvolusi Linier**!
\end{enumerate}
\end{penurunanbox}

\subsection{Sinyal Deterministik vs Sinyal Acak (Random / Stokastik)}

\begin{center}
  \includegraphics[width=0.95\linewidth]{assets/sinyal_deterministik_vs_acak.png}
  \captionof{figure}{Perbandingan sifat analitis sinyal Deterministik (matematis eksak) versus Sinyal Acak / Stokastik (probabilistik).}
\end{center}

\begin{penurunanbox}[Penurunan Teorema Wiener-Khinchin: Dari Autokorelasi ke Spektrum Daya (PSD)]
Mengapa Transformasi Fourier dari Autokorelasi $R_{xx}[k]$ menghasilkan Kerapatan Daya Spektral $S_{xx}(\omega)$?
\begin{enumerate}[leftmargin=*]
  \item \textbf{Definisi Transformasi Fourier Sampel Berhingga $N$:} $X_N(\omega) = \sum_{n=0}^{N-1} x[n] e^{-j\omega n}$.
  \item \textbf{Perhitungan Daya Spektral Rata-rata:}
  \[
  S_{xx}(\omega) = \lim_{N \to \infty} \frac{1}{N} \mathbb{E}\left[ |X_N(\omega)|^2 \right] = \lim_{N \to \infty} \frac{1}{N} \mathbb{E}\left[ \left( \sum_{n=0}^{N-1} x[n] e^{-j\omega n} \right) \left( \sum_{m=0}^{N-1} x[m] e^{j\omega m} \right) \right]
  \]
  \item \textbf{Gabungkan Penjumlahan Ganda:}
  \[
  S_{xx}(\omega) = \lim_{N \to \infty} \frac{1}{N} \sum_{n=0}^{N-1} \sum_{m=0}^{N-1} \mathbb{E}[x[n]x[m]] e^{-j\omega(n-m)}
  \]
  \item \textbf{Substitusi Variabel Selisih Waktu $k = n - m$:} Untuk proses stasioner lebar (WSS), $\mathbb{E}[x[n]x[m]] = R_{xx}[n-m] = R_{xx}[k]$. Penjumlahan ganda menyusut menjadi Transformasi Fourier Diskrit:
  \begin{equation}
  \mathbf{S_{xx}(\omega) = \sum_{k=-\infty}^{\infty} R_{xx}[k] e^{-j\omega k}}
  \end{equation}
\end{enumerate}
\end{penurunanbox}

\subsection{Perbandingan Domain Frekuensi: Sinyal Waktu Kontinu vs Sinyal Waktu Diskrit}

\begin{center}
  \includegraphics[width=0.95\linewidth]{assets/frekuensi_kontinu_vs_diskrit.png}
  \captionof{figure}{Pemetaan fundamental domain frekuensi fisik kontinu ($-\infty < \Omega < \infty$) ke dalam domain frekuensi diskrit terbatas periodik ($-\pi \leq \omega \leq \pi$).}
\end{center}

\begin{penurunanbox}[Asal-Usul Periodisitas Spektrum DTFT $X(e^{j(\omega + 2\pi)}) = X(e^{j\omega})$]
Mengapa spektrum frekuensi diskrit dijamin 100\% periodik dengan periode $2\pi$?
\begin{enumerate}[leftmargin=*]
  \item Berdasarkan definisi Transformasi Fourier Waktu Diskrit (DTFT): $X(e^{j\omega}) = \sum_{n=-\infty}^\infty x[n] e^{-j\omega n}$.
  \item Evaluasi nilai DTFT pada frekuensi tergeser $\omega + 2\pi k$:
  \[
  X(e^{j(\omega + 2\pi k)}) = \sum_{n=-\infty}^\infty x[n] e^{-j(\omega + 2\pi k)n} = \sum_{n=-\infty}^\infty x[n] e^{-j\omega n} \cdot e^{-j 2\pi k n}
  \]
  \item Berdasarkan rumus identitas Euler: $e^{-j 2\pi k n} = \cos(2\pi kn) - j \sin(2\pi kn)$. Karena $k \in \mathbb{Z}$ dan $n \in \mathbb{Z}$, hasil kali $kn$ selalu bilangan bulat, sehingga $\cos(2\pi kn) = 1$ dan $\sin(2\pi kn) = 0 \implies e^{-j 2\pi k n} = 1$.
  \item Maka:
  \begin{equation}
  \mathbf{X(e^{j(\omega + 2\pi k)}) = \sum_{n=-\infty}^\infty x[n] e^{-j\omega n} \cdot 1 = X(e^{j\omega})} \quad \forall k \in \mathbb{Z}
  \end{equation}
\end{enumerate}
\end{penurunanbox}

\newpage

\subsection{Sinyal Sinusoidal Waktu-Diskrit: Syarat Wajib Periodisitas Bilangan Rasional}

\begin{center}
  \includegraphics[width=0.92\linewidth]{assets/sinus_diskrit_periodisitas_1.png}
  \captionof{figure}{Komparasi: Panel A Periodik ($f = 1/8 \in \mathbb{Q}, N = 8$) vs Panel B Aperiodik ($f = 1/(2\pi) \notin \mathbb{Q}$, titik sampel tidak pernah berulang).}
\end{center}

\begin{penurunanbox}[Penurunan Syarat Mutlak Bilangan Rasional $f = k/N \in \mathbb{Q}$]
Mengapa sinyal sinusoidal diskrit $x[n] = \cos(2\pi f n)$ tidak selalu periodik?
\begin{enumerate}[leftmargin=*]
  \item \textbf{Syarat Definisi Periodik:} Harus ada bilangan bulat positif $N \in \mathbb{Z}^+$ sedemikian rupa sehingga $x[n + N] = x[n]$ untuk semua $n$.
  \item Substitusikan ke dalam fungsi cosinus:
  \[
  \cos(2\pi f (n + N)) = \cos(2\pi f n + 2\pi f N) = \cos(2\pi f n)
  \]
  \item Fungsi cosinus berulang nilainya jika dan hanya jika pergeseran sudutnya merupakan kelipatan bilangan bulat dari satu putaran lingkaran $2\pi k$ ($k \in \mathbb{Z}^+$):
  \[
  2\pi f N = 2\pi k \implies f N = k \implies \mathbf{f = \frac{k}{N}}
  \]
  \item Karena $k \in \mathbb{Z}^+$ dan $N \in \mathbb{Z}^+$, maka rasio $k/N$ menurut definisi aljabar bilangan adalah **Bilangan Rasional ($\mathbb{Q}$)**. Jika frekuensi $f$ memuat bilangan irasional (seperti $\pi$ atau $\sqrt{2}$), persamaan ini mustahil dipenuhi oleh sembarang bilangan bulat $N$, sehingga sinyal dijamin **Aperiodik**.
\end{enumerate}
\end{penurunanbox}

\vspace{0.3cm}
\subsection{Sinyal Sinusoidal Waktu-Diskrit: Fenomena Frekuensi Identik Kelipatan \texorpdfstring{$2\pi$}{2pi}}

\begin{center}
  \includegraphics[width=0.92\linewidth]{assets/sinus_diskrit_identik_2pi_2.png}
  \captionof{figure}{Bukti identitas visual: Frekuensi sudut $\omega_0 = \pi/6$ ($30^\circ$) dan $\omega_1 = 13\pi/6 = \pi/6 + 2\pi$ ($390^\circ$) menghasilkan titik cuplikan yang menempel sama persis 100\%.}
\end{center}

\begin{penurunanbox}[Penurunan Rumus Identitas Modulo $2\pi$ Trigonometri]
Buktikan $\cos((\omega + 2\pi k)n + \theta) = \cos(\omega n + \theta)$:
\begin{enumerate}[leftmargin=*]
  \item Gunakan rumus trigonometri penjumlahan sudut $\cos(A + B) = \cos A \cos B - \sin A \sin B$, dengan $A = \omega n + \theta$ dan $B = 2\pi k n$:
  \[
  \cos((\omega n + \theta) + 2\pi k n) = \cos(\omega n + \theta)\cos(2\pi kn) - \sin(\omega n + \theta)\sin(2\pi kn)
  \]
  \item Karena $k \in \mathbb{Z}$ dan $n \in \mathbb{Z}$, maka $kn$ adalah bilangan bulat, menghasilkan $\cos(2\pi kn) = 1$ dan $\sin(2\pi kn) = 0$.
  \item Maka terbukti secara mutlak:
  \begin{equation}
  \mathbf{\cos((\omega + 2\pi k)n + \theta) = \cos(\omega n + \theta) \cdot 1 - \sin(\omega n + \theta) \cdot 0 = \cos(\omega n + \theta)}
  \end{equation}
\end{enumerate}
\end{penurunanbox}

  \[

\subsection{Spesifikasi \& Parameter Matematis Sistem}

\begin{enumerate}[leftmargin=*]
  \item \textbf{Rantai Fisik \& Sensorik (Subbab 1.1):}  
  Tiga elektroda sensor bio-potensial $Ag/AgCl$ ($M = 3$ kanal) merekam sinyal elektrofisiologis kontinu pasien yang terkontaminasi oleh interferensi derau jala-jala listrik $50\text{ Hz}$ dan osilasi pernapasan respirasi lambat $0.2\text{ Hz}$:
  \begin{align}
  x_1(t) &= 3.0 \cos(2\pi \cdot 10 t) + 1.2 \cos(2\pi \cdot 50 t) + 0.8 \sin(2\pi \cdot 0.2 t) \quad [\text{Volt}] \\
  x_2(t) &= 2.5 \cos(2\pi \cdot 10 t - \pi/6) + 1.2 \cos(2\pi \cdot 50 t) + 0.4 \sin(2\pi \cdot 0.2 t) \quad [\text{Volt}] \\
  x_3(t) &= 1.8 \cos(2\pi \cdot 10 t + \pi/4) + 1.2 \cos(2\pi \cdot 50 t) - 0.5 \sin(2\pi \cdot 0.2 t) \quad [\text{Volt}]
  \end{align}

  \item \textbf{Struktur Data Multikanal (Subbab 1.3):}  
  Sinyal multikanal diorganisasikan ke dalam matriks spasio-temporal $\mathbf{X}(t) \in \mathbb{R}^{3 \times N}$. Dilakukan operasi pembobotan spasial diferensial untuk membatalkan komponen *common-mode noise* $50\text{ Hz}$ dengan vektor bobot $\mathbf{w} = [1, -1, 0]^T$:
  \begin{equation}
  s_a(t) = \mathbf{w}^T \mathbf{x}(t) = x_1(t) - x_2(t)
  \end{equation}

  \item \textbf{Spesifikasi ADC \& Digitalisasi (Subbab 1.2 \& 1.4):}
  \begin{itemize}
    \item Frekuensi Sampling Sistem: $F_s = 200\text{ Hz}$ ($T_s = 5\text{ ms} = 0.005\text{ s}$).
    \item Terdapat komponen derau interferensi frekuensi tinggi liar tak terfilter pada $f_{\text{noise}} = 230\text{ Hz}$ dengan persamaan $v_{\text{noise}}(t) = 0.5 \cos(2\pi \cdot 230 t)$.
    \item Konverter ADC Uniform 4-Bit ($B = 4\text{ bit} \implies L = 16\text{ level}$), dengan rentang tegangan dinamis $V_{\text{min}} = -4.0\text{ V}$ hingga $V_{\text{maks}} = +4.0\text{ V}$ ($V_{\text{range}} = 8.0\text{ V}$).
  \end{itemize}

  \item \textbf{Filter Digital Pemroses LTI (Subbab 1.1 \& 1.3):}  
  Sinyal tercuplik $s[n]$ dilewatkan ke sebuah filter rata-rata bergerak 3-titik (*3-Point Moving Average Filter*):
  \begin{equation}
  y[n] = \frac{1}{3}s[n] + \frac{1}{3}s[n-1] + \frac{1}{3}s[n-2]
  \end{equation}
\end{enumerate}

\subsection{Penyelesaian Komputasi Analitis Langkah-demi-Langkah}

\subsubsection{Langkah 1: Reduksi Multikanal \& Analisis Parameter Gelombang (Subbab 1.1 \& 1.3)}
Hitung sinyal hasil kombinasi spasial $s_a(t) = x_1(t) - x_2(t)$:
\begin{align*}
s_a(t) &= \left[ 3.0 \cos(20\pi t) + 1.2 \cos(100\pi t) + 0.8 \sin(0.4\pi t) \right] \\
&\quad - \left[ 2.5 \cos(20\pi t - \pi/6) + 1.2 \cos(100\pi t) + 0.4 \sin(0.4\pi t) \right] \\
&= \underbrace{\left[ 3.0 \cos(20\pi t) - 2.5 \cos(20\pi t - \pi/6) \right]}_{\text{Komponen Jantung 10 Hz}} + \underbrace{[1.2 - 1.2]\cos(100\pi t)}_{\mathbf{0\text{ (Derau 50 Hz Lenyap!)}}} + \underbrace{0.4 \sin(0.4\pi t)}_{\text{Respirasi 0.2 Hz}}
\end{align*}
Gunakan identitas trigonometri fasor untuk menyederhanakan komponen $10\text{ Hz}$:
\begin{align*}
2.5 \cos(20\pi t - \pi/6) &= 2.5 \left[ \cos(20\pi t)\cos(\pi/6) + \sin(20\pi t)\sin(\pi/6) \right] \\
&= 2.5 \left[ \frac{\sqrt{3}}{2} \cos(20\pi t) + \frac{1}{2} \sin(20\pi t) \right] \approx 2.165 \cos(20\pi t) + 1.250 \sin(20\pi t)
\end{align*}
Maka:
\begin{equation}
s_{10\text{Hz}}(t) = (3.0 - 2.165)\cos(20\pi t) - 1.250 \sin(20\pi t) = 0.835 \cos(20\pi t) - 1.250 \sin(20\pi t)
\end{equation}
Amplitudo resultan $A_R$ dan sudut fase $\phi_R$:
\begin{equation}
A_R = \sqrt{(0.835)^2 + (-1.250)^2} = \sqrt{0.6972 + 1.5625} = \sqrt{2.2597} \approx \mathbf{1.503\text{ Volt}}
\end{equation}
\begin{equation}
\phi_R = \operatorname{atan2}(-1.250, 0.835) \approx -56.27^\circ \approx -0.982\text{ rad}
\end{equation}
Persamaan analitis bersih sinyal kontinu yang masuk ke ADC (termasuk derau liar $230\text{ Hz}$):
\begin{equation}
s_{\text{total}}(t) = 1.503 \cos(2\pi \cdot 10 t - 0.982) + 0.4 \sin(2\pi \cdot 0.2 t) + 0.5 \cos(2\pi \cdot 230 t)
\end{equation}

\subsubsection{Langkah 2: Evaluasi Nyquist, Identifikasi Aliasing, dan Pemetaan Frekuensi Diskrit (Subbab 1.2 \& 1.5)}
\begin{enumerate}[leftmargin=*]
  \item \textbf{Analisis Frekuensi Nyquist:}  
  Frekuensi sampling $F_s = 200\text{ Hz} \implies$ Frekuensi Nyquist ($f_{\text{fold}}$) adalah:
  \begin{equation}
  f_{\text{fold}} = \frac{F_s}{2} = \frac{200\text{ Hz}}{2} = \mathbf{100\text{ Hz}}
  \end{equation}
  \item \textbf{Evaluasi Komponen $F_1 = 10\text{ Hz}$ dan $F_2 = 0.2\text{ Hz}$:}  
  Karena $F_1, F_2 < 100\text{ Hz}$, kedua komponen berada di dalam pita Nyquist dan disampling secara aman bebas aliasing. Frekuensi digitalnya:
  \begin{equation}
  \omega_1 = \frac{2\pi F_1}{F_s} = \frac{2\pi \cdot 10}{200} = \mathbf{\frac{\pi}{10}\ \text{rad/sampel}}, \qquad f_1 = \frac{10}{200} = \mathbf{\frac{1}{20}\ \text{siklus/sampel}}
  \end{equation}
  \begin{equation}
  \omega_2 = \frac{2\pi F_2}{F_s} = \frac{2\pi \cdot 0.2}{200} = \mathbf{\frac{\pi}{500}\ \text{rad/sampel}}, \qquad f_2 = \frac{0.2}{200} = \mathbf{\frac{1}{1000}\ \text{siklus/sampel}}
  \end{equation}
  
  \item \textbf{Evaluasi Aliasing Komponen Liar $F_3 = 230\text{ Hz}$:}  
  Karena $F_3 = 230\text{ Hz} > 100\text{ Hz}$, terjadi pelanggaran kriteria Nyquist (*Under-sampling*). Komponen ini terlipat ke frekuensi semu alias:
  \begin{equation}
  f_{\text{alias}} = |F_3 - 1 \cdot F_s| = |230\text{ Hz} - 200\text{ Hz}| = \mathbf{30\text{ Hz}}
  \end{equation}
  Frekuensi sudut diskrit aliasnya:
  \begin{equation}
  \omega_{\text{alias}} = \frac{2\pi \cdot 30}{200} = \mathbf{\frac{3\pi}{10}\ \text{rad/sampel}} \quad \left(f_{\text{alias}} = \frac{3}{20}\right)
  \end{equation}

  \item \textbf{Uji Periodisitas Diskrit (Subbab 1.5.4):}
  \begin{itemize}
    \item Untuk $\omega_1$: $f_1 = 1/20 \in \mathbb{Q} \implies$ **Periodik** dengan periode fundamental $N_1 = 20\text{ sampel}$.
    \item Untuk $\omega_2$: $f_2 = 1/1000 \in \mathbb{Q} \implies$ **Periodik** dengan periode fundamental $N_2 = 1000\text{ sampel}$.
    \item Untuk $\omega_{\text{alias}}$: $f_{\text{alias}} = 3/20 \in \mathbb{Q} \implies$ **Periodik** dengan periode fundamental $N_3 = \frac{20}{\gcd(3,20)} = 20\text{ sampel}$.
    \item **Periode Fundamental Total Sinyal Campuran:**
    \begin{equation}
    N_{\text{total}} = \operatorname{KPK}(N_1, N_2, N_3) = \operatorname{KPK}(20, 1000, 20) = \mathbf{1000\text{ sampel}} \quad (\text{setara } 5.0\text{ detik})
    \end{equation}
  \end{itemize}
\end{enumerate}

\subsubsection{Langkah 4: Pemrosesan Filter LTI \& Uji Karakteristik Sistem (Subbab 1.1 \& 1.3)}
Sinyal terkuantisasi $s_q[n]$ diproses oleh filter rata-rata bergerak 3-titik:
\begin{equation}
y[n] = \frac{1}{3} s_q[n] + \frac{1}{3} s_q[n-1] + \frac{1}{3} s_q[n-2]
\end{equation}

\textbf{1. Verifikasi 4 Sifat Karakteristik Sistem Filter:}
\begin{itemize}[leftmargin=*]
  \item \textbf{Uji Linearitas:} Misalkan $s_q[n] = a x_1[n] + b x_2[n]$:
  \begin{align*}
  y[n] &= \frac{1}{3}(a x_1[n] + b x_2[n]) + \frac{1}{3}(a x_1[n-1] + b x_2[n-1]) + \frac{1}{3}(a x_1[n-2] + b x_2[n-2]) \\
  &= a \left[\frac{1}{3}x_1[n] + \frac{1}{3}x_1[n-1] + \frac{1}{3}x_1[n-2]\right] + b \left[\frac{1}{3}x_2[n] + \frac{1}{3}x_2[n-1] + \frac{1}{3}x_2[n-2]\right] \\
  &= a y_1[n] + b y_2[n] \quad \mathbf{(\text{Terbukti Linier!})}
  \end{align*}
  \item \textbf{Uji Time-Invariance:} Jika input digeser $s_q[n-n_0]$, output menjadi $\frac{1}{3}\sum_{k=0}^2 s_q[n-n_0-k] = y[n-n_0] \quad \mathbf{(\text{Terbukti Time-Invariant!})}$.
  \item \textbf{Uji Kausalitas:} Respon impuls sistem adalah $h[n] = \frac{1}{3}\delta[n] + \frac{1}{3}\delta[n-1] + \frac{1}{3}\delta[n-2]$. Karena $h[n] = 0$ untuk seluruh $n < 0$, maka sistem \textbf{Terbukti Kausal}.
  \item \textbf{Uji Stabilitas BIBO:}
  \begin{equation}
  S_h = \sum_{n=-\infty}^{\infty} |h[n]| = |h[0]| + |h[1]| + |h[2]| = \frac{1}{3} + \frac{1}{3} + \frac{1}{3} = \mathbf{1.0 < \infty} \quad \mathbf{(\text{Terbukti Stabil BIBO!})}
  \end{equation}
\end{itemize}

\vspace{0.2cm}
\textbf{2. Perhitungan Respon Frekuensi Filter terhadap Derau Alias 30 Hz ($3\pi/10$):}
\begin{equation}
H(e^{j\omega}) = \sum_{n=0}^2 h[n] e^{-j\omega n} = \frac{1}{3}\left( 1 + e^{-j\omega} + e^{-j2\omega} \right) = \frac{1}{3} e^{-j\omega} \left( 1 + 2\cos(\omega) \right)
\end{equation}
Magnitudo penguatan filter $|H(e^{j\omega})| = \frac{1}{3}|1 + 2\cos(\omega)|$.
\begin{itemize}[leftmargin=*]
  \item Pada sinyal jantung $\omega_1 = \frac{\pi}{10} = 18^\circ$:
  \begin{equation}
  |H(e^{j\pi/10})| = \frac{1}{3}|1 + 2\cos(18^\circ)| = \frac{1}{3}|1 + 2(0.9511)| \approx \mathbf{0.9674\ (-0.29\text{ dB})} \implies \text{Lolos Utuh!}
  \end{equation}
  \item Pada derau liar alias $\omega_{\text{alias}} = \frac{3\pi}{10} = 54^\circ$:
  \begin{equation}
  |H(e^{j3\pi/10})| = \frac{1}{3}|1 + 2\cos(54^\circ)| = \frac{1}{3}|1 + 2(0.5878)| \approx \mathbf{0.7252\ (-2.79\text{ dB})} \implies \text{Tereduksi!}
  \end{equation}
\end{itemize}

\vspace{0.2cm}
\textbf{3. Kalkulasi Numerik Keluaran Filter $y[2]$:}
Dengan asumsi kondisi diam awal $s_q[-1] = 0, s_q[-2] = 0$:
\begin{equation}
y[2] = \frac{1}{3}s_q[2] + \frac{1}{3}s_q[1] + \frac{1}{3}s_q[0] = \frac{1}{3}(1.250) + \frac{1}{3}(1.250) + \frac{1}{3}(1.250) = \mathbf{1.250\text{ Volt}}
\end{equation}

\vspace{0.3cm}
\begin{tcolorbox}[colback=accent!10, colframe=accent, arc=2mm, title=\textbf{Kesimpulan Grand Synthesis Subbab 1.1 - 1.5}]
Studi kasus ini secara matematis membuktikan integrasi end-to-end:
\begin{enumerate}
  \item \textbf{Subbab 1.1:} Pemodelan transduksi bio-potensial, derau termal, dan eliminasi interferensi fasor fase kuadratur.
  \item \textbf{Subbab 1.2:} Laju sampling $F_s$, pencegahan aliasing Nyquist, serta kuantisasi uniform $4$-bit dengan presisi $\Delta = 0.5\text{ V}$.
  \item \textbf{Subbab 1.3:} Aljabar linear multikanal matriks spasio-temporal dan dekomposisi sinyal elementer.
  \item \textbf{Subbab 1.4:} Topologi sinyal digital kuadran-4 dan model derau kuantisasi aditif berdaya $\sigma_e^2 = \Delta^2/12$.
  \item \textbf{Subbab 1.5:} Periodisitas rasional frekuensi diskrit $f = k/N$, identitas kesamaan modulo $2\pi$, dan penapisan magnitudo filter LTI.
\end{enumerate}
\end{tcolorbox}

\subsection{Anatomi \& Parameter Gelombang}
\begin{table}[h!]

\textbf{Amplitudo ($A$)} & Simpangan puncak ekstrem gelombang dari titik nol, menentukan daya rata-rata $P = A^2/2$ dan tegangan efektif $V_{\text{rms}} = A/\sqrt{2}$. \\
\textbf{Frekuensi ($f, F$)} & Jumlah osilasi siklus gelombang lengkap dalam 1 detik ($\text{Hertz} = \text{s}^{-1}$). \\
\textbf{Periode ($T$)} & Interval waktu minimum yang diperlukan untuk menyelesaikan 1 siklus osilasi periodik penuh ($T = 1/f$). \\
\textbf{Frekuensi Sudut ($\Omega, \omega$)} & Laju kecepatan rotasi fase gelombang ($\Omega = 2\pi F\text{ rad/s}$ untuk kontinu; $\omega = 2\pi f\text{ rad/sampel}$ untuk diskrit). \\
\textbf{Fase Awal ($\phi, \theta$)} & Posisi sudut mula gelombang pada $t = 0$, setara dengan pergeseran translasi waktu $\Delta t = -\phi/\Omega$. \\
\bottomrule
\end{tabularx}

\subsection{Proses Digitalisasi \& ADC}
\begin{table}[h!]
\centering
\small
\begin{tabularx}{\textwidth}{l>{\raggedright\arraybackslash}X}
\toprule
\textbf{Istilah} & \textbf{Definisi \& Konsep Kunci} \\
\midrule
\textbf{ADC (\textit{A/D Converter})} & Subsistem pengubah sinyal kontinu analog menjadi kata-kata biner digital melalui rantai sampling, kuantisasi, dan encoding. \\
\textbf{Pencuplikan (\textit{Sampling})} & Proses modulasi sinyal kontinu dengan deretan pulsa Dirac pada interval waktu berkala $T_s = 1/F_s$. \\
\textbf{Teorema Nyquist} & Syarat batas absolut bebas aliasing: frekuensi sampling wajib memenuhi $F_s \ge 2 f_{\max}$. \\
\textbf{Aliasing} & Distorsi lipatan spektral di mana frekuensi di atas $F_s/2$ menyamar menjadi frekuensi rendah palsu $f_{\text{alias}} = |F - k F_s|$. \\
\textbf{Kuantisasi (\textit{Quantization})} & Pembulatan non-linier amplitudo kontinu ke $L = 2^B$ level diskrit terdekat dengan lebar step $\Delta = V_{\text{range}}/2^B$. \\
\textbf{SQNR} & Rasio Daya Sinyal terhadap Derau Kuantisasi berdaya $\sigma_e^2 = \Delta^2/12$, bernilai $\text{SQNR} \approx 6.02 B + 1.76\text{ dB}$. \\
